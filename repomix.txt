package httpx

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"net/url"
	"time"
)

type Client struct {
	token   string
	baseURL string
	http    *http.Client
}

func New(token, baseURL string) *Client {
	return &Client{
		token:   token,
		baseURL: baseURL,
		http:    &http.Client{Timeout: 60 * time.Second},
	}
}

func (c *Client) do(req *http.Request) (*http.Response, error) {
	req.Header.Set("Authorization", "Bearer "+c.token)
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("User-Agent", "deepseek-devkit/0.1")
	// при необходимости добавьте x-client-* заголовки
	return c.http.Do(req)
}

func (c *Client) GetJSON(ctx context.Context, path string, out interface{}) error {
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, c.baseURL+path, nil)
	if err != nil {
		return err
	}
	resp, err := c.do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode < 200 || resp.StatusCode >= 300 {
		body, _ := io.ReadAll(resp.Body)
		return fmt.Errorf("http %d: %s", resp.StatusCode, body)
	}
	return json.NewDecoder(resp.Body).Decode(out)
}

func (c *Client) GetJSONWithQuery(
	ctx context.Context,
	path string,
	query map[string]string,
	out interface{},
) error {
	u, err := url.Parse(c.baseURL + path)
	if err != nil {
		return err
	}
	q := u.Query()
	for k, v := range query {
		q.Set(k, v)
	}
	u.RawQuery = q.Encode()

	req, err := http.NewRequestWithContext(ctx, http.MethodGet, u.String(), nil)
	if err != nil {
		return err
	}
	resp, err := c.do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode < 200 || resp.StatusCode >= 300 {
		body, _ := io.ReadAll(resp.Body)
		return fmt.Errorf("http %d: %s", resp.StatusCode, body)
	}
	return json.NewDecoder(resp.Body).Decode(out)
}

func (c *Client) PostJSON(
	ctx context.Context,
	path string,
	body interface{},
	out interface{},
) error {
	b, err := json.Marshal(body)
	if err != nil {
		return err
	}
	req, err := http.NewRequestWithContext(ctx, http.MethodPost, c.baseURL+path, bytes.NewReader(b))
	if err != nil {
		return err
	}
	resp, err := c.do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode < 200 || resp.StatusCode >= 300 {
		raw, _ := io.ReadAll(resp.Body)
		return fmt.Errorf("http %d: %s", resp.StatusCode, raw)
	}
	if out != nil {
		return json.NewDecoder(resp.Body).Decode(out)
	}
	return nil
}

// PostJSONStream отправляет запрос с PoW-заголовком и возвращает потоковый ответ.
func (c *Client) PostJSONStream(
	ctx context.Context,
	path string,
	body interface{},
	powHeader string,
) (*http.Response, error) {
	b, err := json.Marshal(body)
	if err != nil {
		return nil, err
	}
	req, err := http.NewRequestWithContext(ctx, http.MethodPost, c.baseURL+path, bytes.NewReader(b))
	if err != nil {
		return nil, err
	}
	req.Header.Set("x-ds-pow-response", powHeader)
	// Таймаут для стриминга должен быть больше или отсутствовать;
	// здесь используется клиент с общим таймаутом.
	resp, err := c.do(req)
	if err != nil {
		return nil, err
	}
	if resp.StatusCode < 200 || resp.StatusCode >= 300 {
		raw, _ := io.ReadAll(resp.Body)
		resp.Body.Close()
		return nil, fmt.Errorf("http %d: %s", resp.StatusCode, raw)
	}
	return resp, nil
}

import (
    "context"
    "fmt"
    "net/http"
    "time"
    "sync"

    "github.com/ice-rider/deepseek-devkit/internal/httpx"
)

const defaultBaseURL = "https://chat.deepseek.com"

// devkit.go
type Client struct {
	token     string
	baseURL   string
	http      *http.Client
	hx        *httpx.Client
	lastMsgID map[string]string
	msgMu     sync.Mutex
}

func NewClient(token string) *Client {
	return &Client{
		token:     token,
		baseURL:   defaultBaseURL,
		http:      &http.Client{Timeout: 60 * time.Second},
		lastMsgID: make(map[string]string),
	}
}

func (c *Client) getLastMsgID(chatID string) string {
	c.msgMu.Lock()
	defer c.msgMu.Unlock()
	return c.lastMsgID[chatID]
}

func (c *Client) setLastMsgID(chatID, id string) {
	if id == "" {
		return
	}
	c.msgMu.Lock()
	defer c.msgMu.Unlock()
	c.lastMsgID[chatID] = id
}

// Authorize проверяет токен и инициализирует внутренний HTTP-клиент.
// Если токен пуст, возвращается ошибка.
func (c *Client) Authorize(ctx context.Context) (*User, error) {
    if c.token == "" {
        return nil, fmt.Errorf("devkit: token is empty")
    }
    c.hx = httpx.New(c.token, c.baseURL)

    var raw struct {
        Data struct {
            BizData struct {
                User User `json:"user"`
            } `json:"biz_data"`
        } `json:"data"`
    }

    if err := c.hx.GetJSON(ctx, "/api/v0/users/current", &raw); err != nil {
        return nil, fmt.Errorf("authorize: %w", err)
    }
    raw.Data.BizData.User.Token = c.token
    return &raw.Data.BizData.User, nil
}

// GetCurrentUser — псевдоним для Authorize, возвращает информацию о пользователе.
func (c *Client) GetCurrentUser(ctx context.Context) (*User, error) {
    return c.Authorize(ctx)
}

package devkit

import (
	"bufio"
	"context"
	"encoding/json"
	"fmt"
	"strconv"
	"strings"
)

// NewMessage отправляет сообщение в чат.
func (c *Client) NewMessage(
	ctx context.Context,
	chatID, prompt string,
	stream bool,
) (*NewMessageResult, <-chan string, error) {

	powHeader, err := c.preparePoW(ctx, "/api/v0/chat/completion")
	if err != nil {
		return nil, nil, err
	}

	// Гарантируем, что мы знаем последний message_id в этой сессии.
	if c.getLastMsgID(chatID) == "" {
		c.refreshLastMsgID(ctx, chatID)
	}

	var parentID interface{} // именно interface{}, чтобы отправить nil или int
	if idStr := c.getLastMsgID(chatID); idStr != "" {
		// DeepSeek ждет integer, конвертируем строку обратно в число
		if idInt, err := strconv.Atoi(idStr); err == nil {
			parentID = idInt
		} else {
			parentID = idStr // fallback
		}
	}

	body := map[string]interface{}{
		"chat_session_id":   chatID,
		"parent_message_id": parentID, // <-- Теперь это число!
		"model_type":        "default",
		"prompt":            prompt,
		"ref_file_ids":      []string{},
		"thinking_enabled":  false,
		"search_enabled":    false,
	}

	if stream {
		ch := make(chan string, 64)
		go func() {
			defer close(ch)
			_ = c.streamCompletion(ctx, chatID, body, powHeader, ch)
			// Fallback: если SSE не дал response_message_id — добираем из истории.
			if c.getLastMsgID(chatID) == "" {
				c.refreshLastMsgID(context.Background(), chatID)
			}
		}()
		return nil, ch, nil
	}

	res, err := c.syncCompletion(ctx, chatID, body, powHeader, prompt)
	if err != nil {
		return nil, nil, err
	}
	return res, nil, nil
}

// syncCompletion читает SSE и собирает полный ответ.
func (c *Client) syncCompletion(
	ctx context.Context,
	chatID string,
	body map[string]interface{},
	powHeader string,
	prompt string,
) (*NewMessageResult, error) {

	resp, err := c.hx.PostJSONStream(ctx, "/api/v0/chat/completion", body, powHeader)
	if err != nil {
		return nil, fmt.Errorf("completion: %w", err)
	}
	defer resp.Body.Close()

	var assistant strings.Builder
	var userMsgID, assistMsgID string

	scanner := bufio.NewScanner(resp.Body)
	scanner.Buffer(make([]byte, 1<<20), 1<<20)
	for scanner.Scan() {
		line := scanner.Text()
		if !strings.HasPrefix(line, "data:") {
			continue
		}
		data := strings.TrimSpace(strings.TrimPrefix(line, "data:"))
		if data == "" || data == "[DONE]" {
			continue
		}

		var evt map[string]interface{}
		if err := json.Unmarshal([]byte(data), &evt); err != nil {
			continue
		}

		if id, ok := evt["request_message_id"].(string); ok && id != "" {
			userMsgID = id
		}
		
		// event: ready приходит отдельным пакетом, но мы парсим все data:
		if idFloat, ok := evt["response_message_id"].(float64); ok {
			c.setLastMsgID(chatID, strconv.Itoa(int(idFloat)))
		}

		// Первый чанк ответа содержит вложенный объект response
		if v, ok := evt["v"].(map[string]interface{}); ok {
			if resp, ok := v["response"].(map[string]interface{}); ok {
				if idFloat, ok := resp["message_id"].(float64); ok {
					c.setLastMsgID(chatID, strconv.Itoa(int(idFloat)))
				}
			}
		}

		if txt, ok := extractContent(evt); ok {
			assistant.WriteString(txt)
		}
	}
	if err := scanner.Err(); err != nil {
		return nil, err
	}

	if assistMsgID != "" {
		c.setLastMsgID(chatID, assistMsgID)
	}

	// Fallback: если SSE не дал response_message_id, берём из истории.
	if c.getLastMsgID(chatID) == "" {
		c.refreshLastMsgID(ctx, chatID)
	}

	return &NewMessageResult{
		UserMessage:      &Message{ID: userMsgID, Role: "user", Content: prompt},
		AssistantMessage: &Message{ID: assistMsgID, Role: "assistant", Content: assistant.String()},
	}, nil
}

// streamCompletion читает SSE и отправляет текстовые дельты в канал.
func (c *Client) streamCompletion(
	ctx context.Context,
	chatID string,
	body map[string]interface{},
	powHeader string,
	out chan<- string,
) error {

	resp, err := c.hx.PostJSONStream(ctx, "/api/v0/chat/completion", body, powHeader)
	if err != nil {
		return err
	}
	defer resp.Body.Close()

	scanner := bufio.NewScanner(resp.Body)
	scanner.Buffer(make([]byte, 1<<20), 1<<20)
	for scanner.Scan() {
		select {
		case <-ctx.Done():
			return ctx.Err()
		default:
		}

		line := scanner.Text()
		if !strings.HasPrefix(line, "data:") {
			continue
		}
		data := strings.TrimSpace(strings.TrimPrefix(line, "data:"))
		if data == "" || data == "[DONE]" {
			continue
		}

		var evt map[string]interface{}
		if err := json.Unmarshal([]byte(data), &evt); err != nil {
			continue
		}

		if id, ok := evt["response_message_id"].(string); ok && id != "" {
			c.setLastMsgID(chatID, id)
		}

		if txt, ok := extractContent(evt); ok {
			out <- txt
		}
	}
	return scanner.Err()
}

// extractContent достаёт текстовый кусок из события SSE.
//
// DeepSeek-веб присылает куски в поле "v". Поле "p" (path) говорит, к чему
// относится "v". Контентом считаем:
//   - p отсутствует/пуст        -> обычный чанк ответа
//   - p заканчивается на /content -> явный чанк ответа
// Пропускаем пути вида "response/thinking_content", "response/status" и т.п.
func extractContent(evt map[string]interface{}) (string, bool) {
	if p, ok := evt["p"].(string); ok && p != "" && !strings.HasSuffix(p, "/content") {
		return "", false
	}
	v, ok := evt["v"].(string)
	if !ok || v == "" {
		return "", false
	}
	return v, true
}

// StopGeneration останавливает генерацию ответа в указанной сессии.
//
// DeepSeek требует два идентификатора: chat_session_id и message_id,
// причём message_id — это идентификатор ответа ассистента
// (response_message_id из SSE), который ещё генерируется. Он хранится
// в Client.lastMsgID и обновляется на каждом событии стрима/синка.
//
// Если в сессии ещё не было ответов — вернёт ошибку: останавливать нечего.
// Метод не блокирует вызывающего: если нужно прервать текущий
// syncCompletion/streamCompletion, вызывайте из отдельной горутины.
func (c *Client) StopGeneration(ctx context.Context, chatID string) error {
	if chatID == "" {
		return fmt.Errorf("devkit: chatID is empty")
	}

	msgID := c.getLastMsgID(chatID)
	if msgID == "" {
		return fmt.Errorf("devkit: no active message in chat %q", chatID)
	}

	body := map[string]string{
		"chat_session_id": chatID,
		"message_id":      msgID,
	}

	if err := c.hx.PostJSON(ctx, "/api/v0/chat/stop_stream", body, nil); err != nil {
		return fmt.Errorf("stop generation: %w", err)
	}
	return nil
}

// refreshLastMsgID подтягивает последний assistant message_id из истории
// и запоминает его как parent для следующего хода. Стоит один GET на вызов,
// зато гарантирует, что диалог не «разорвётся» на ветки.
func (c *Client) refreshLastMsgID(ctx context.Context, chatID string) {
	msgs, err := c.GetChatById(ctx, chatID)
	if err != nil {
		return
	}
	for i := len(msgs) - 1; i >= 0; i-- {
		if msgs[i].Role == "assistant" && msgs[i].ID != "" {
			c.setLastMsgID(chatID, msgs[i].ID)
			return
		}
	}
}

import (

// powChallengeResponse — ответ эндпоинта /api/v0/chat/create_pow_challenge.
type powChallengeResponse struct {
	Data struct {
		BizCode int `json:"biz_code"`
		BizData struct {
			Challenge pow.Challenge `json:"challenge"` // <-- вложенный объект
		} `json:"biz_data"`
	} `json:"data"`
}

// preparePoW получает челлендж для targetPath и возвращает готовый заголовок
// x-ds-pow-response.
func (c *Client) preparePoW(ctx context.Context, targetPath string) (string, error) {
	body := map[string]string{"target_path": targetPath}

	var ch powChallengeResponse
	if err := c.hx.PostJSON(ctx, "/api/v0/chat/create_pow_challenge", body, &ch); err != nil {
		return "", fmt.Errorf("pow challenge: %w", err)
	}
	if ch.Data.BizCode != 0 {
		return "", fmt.Errorf("pow challenge: biz_code=%d", ch.Data.BizCode)
	}

	// Теперь передаём контекст и указатель на Challenge — как требует ds2api/pow.
	header, err := pow.SolveAndBuildHeader(ctx, &ch.Data.BizData.Challenge)
	if err != nil {
		return "", fmt.Errorf("pow solve: %w", err)
	}
	return header, nil
}
